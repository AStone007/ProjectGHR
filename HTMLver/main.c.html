1    #include <stdio.h>
2    #include "io.h"
3    #include "wf.h"
4    
5    int main(void) {
6    
7        //load directory of CSV files
8        Directory *dir = load_directory("./data");
9        if (dir == NULL) {
10           printf("Error: Failed to load directory\n");
11           return 1;
12       }
13       //loop through each file
14       for (int i = 0; i < dir->file_count; i++) {
15           File *file = &dir->files[i];
16           printf("Processing file: %s\n", file->filename);
17           //phase calculations
18           for (Phase p = PHASE_A; p <= PHASE_C; p++) {
19               double rms = compute_rms(file, p);
20               double p2p = compute_peak_to_peak(file, p);
21               double dc = compute_dc_offset(file, p);
22               double std = compute_std_dev(file, p);
23               //write results
24               write_result(file, "RMS", rms);
25               write_result(file, "Peak-to-Peak", p2p);
26               write_result(file, "DC Offset", dc);
27               write_result(file, "Standard Deviation", std);
28               //compliance check
29               int compliant = check_compliance(rms);
30               write_result(file, "Voltage Compliance", compliant);
31           }
32           // non-repetitive calculations
33           double pf = power_factor(file);
34           double thd = thd_percent(file);
35           double freq = frequency(file);
36           write_result(file, "Power Factor", pf);
37           write_result(file, "THD (%)", thd);
38           write_result(file, "Frequency (Hz)", freq);
39           //clipping
40           double limit = 325.0; // typical peak for 230V RMS
41           ClippingResult clip = count_clipped(file, limit);
42           write_clipping_result(file, clip);
43       }
44       //clear memory
45       free_directory(dir);
46       printf("complete.\n");
47   
48       return 0;
49   }
