5    double get_voltage(WaveformSample sample, Phase phase) {
6    
7        switch (phase) {
8            case PHASE_A: return sample.phase_A_voltage;
9            case PHASE_B: return sample.phase_B_voltage;
10           case PHASE_C: return sample.phase_C_voltage;
11           default: return 0.0;
12       }
13   }

15   // RMS
16   double compute_rms(File *file, Phase phase) {
17   
18       if (!file || file->num_samples == 0) return 0.0;
19       double sum_square = 0.0;
20       for (int i = 0; i < file->num_samples; i++) {
21           double v = get_voltage(file->samples[i], phase);
22           sum_square += v * v;
23       }
24       return sqrt(sum_square / file->num_samples);
25   }

27   // Peak-to-peak
28   double compute_peak_to_peak(File *file, Phase phase) {
29   
30       if (!file || file->num_samples == 0) return 0.0;
31       double max = get_voltage(file->samples[0], phase);
32                               //max or min may not be 0
33       double min = max;
34       for (int i = 1; i < file->num_samples; i++) {
35           double v = get_voltage(file->samples[i], phase);
36           if (v > max) max = v;
37           if (v < min) min = v;
38       }
39       return max - min;
40   }

42   //dc offset
43   double compute_dc_offset(File *file, Phase phase) {
44   
45       if (!file || file->num_samples == 0) return 0.0;
46       double sum = 0.0;
47       for (int i = 0; i < file->num_samples; i++) {
48           sum += get_voltage(file->samples[i], phase);
49       }
50       return sum / file->num_samples;
51   }

53   //std dev
54   double compute_std_dev(File *file, Phase phase) {
55   
56       if (!file || file->num_samples == 0) return 0.0;
57       double mean = compute_dc_offset(file, phase);
58       double sum = 0.0;
59       for (int i = 0; i < file->num_samples; i++) {
60           double v = get_voltage(file->samples[i], phase);
61           sum += (v - mean) * (v - mean);
62       }
63       return sqrt(sum / file->num_samples);
64   }

66   //clipping
67   ClippingResult count_clipped(File *file, double limit) {
68       ClippingResult result = {0};
69       for (int i = 0; i < file->num_samples; i++) {
70           double t = file->samples[i].timestamp;
71           double Va = get_voltage(file->samples[i], PHASE_A);
72           double Vb = get_voltage(file->samples[i], PHASE_B);
73           double Vc = get_voltage(file->samples[i], PHASE_C);
74           if (fabs(Va) >= limit) {
75               result.times_A[result.count_A++] = t;
76           }
77           if (fabs(Vb) >= limit) {
78               result.times_B[result.count_B++] = t;
79           }
80           if (fabs(Vc) >= limit) {
81               result.times_C[result.count_C++] = t;
82           }
83       }
84       return result;
85   }
